tion, amount, due_date,
                                       client_id, work_order_id, category)
  values (v_wo.company_id, 'income', v_wo.number || ' · atendimento', v_total,
          current_date + p_due_days, v_wo.client_id, p_wo, 'Serviços')
  on conflict (work_order_id) where work_order_id is not null do nothing
  returning id into v_entry;
  if v_entry is null then select id into v_entry from public.financial_entries where work_order_id = p_wo; end if;

  update public.work_orders set billing_entry_id = v_entry, pending_pricing = false where id = p_wo;
  insert into public.work_order_reports(work_order_id, company_id, entry_type, body, author_id)
  values (p_wo, v_wo.company_id, 'history', 'Cobrança gerada em contas a receber', auth.uid());
  return v_entry;
end $$;

-- =====================================================================
-- Criação de empresa em uma única transação: company + owner + assinatura.
-- Nada de user_id/company_id vindos do frontend; só auth.uid().
-- =====================================================================
create or replace function public.zt_create_company(
  p_name text, p_activity text default null, p_has_team boolean default false,
  p_owner_name text default null, p_phone text default null
) returns uuid language plpgsql security definer set search_path = public as $$
declare v_uid uuid := auth.uid(); v_company uuid;
begin
  if v_uid is null then raise exception 'Não autenticado' using errcode = '28000'; end if;
  if p_name is null or length(trim(p_name)) = 0 then raise exception 'Informe o nome da empresa'; end if;

  insert into public.profiles(id, email)
  values (v_uid, (select email from auth.users where id = v_uid))
  on conflict (id) do nothing;

  insert into public.companies(name, activity, has_team, owner_name, phone)
  values (trim(p_name), p_activity, coalesce(p_has_team,false),
          coalesce(p_owner_name, (select full_name from public.profiles where id = v_uid)), p_phone)
  returning id into v_company;

  insert into public.company_members(company_id, user_id, role, status)
  values (v_company, v_uid, 'owner', 'active');

  insert into public.subscriptions(company_id, status, current_period_start, current_period_end)
  values (v_company, 'trial', current_date, current_date + 14)
  on conflict (company_id) do nothing;

  return v_company;
end $$;

  select exists (

$$;

-- filhos da OS acompanham a visibilidade da OS mãe
-- ---------------------------------------------------------------------
-- Filhas da OS · menor privilégio.
-- Owner: controle total. Técnico atribuído: lê a própria OS, acrescenta o
-- que é operacional e não apaga nada — histórico é registro de auditoria.
-- ---------------------------------------------------------------------
create or replace function public.zt_wo_is_mine(w_id uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.work_orders w
    join public.company_members m on m.company_id = w.company_id
    where w.id = w_id and w.assigned_to = auth.uid()
      and m.user_id = auth.uid() and m.status = 'active'
  );
$$;
create or replace function public.zt_wo_is_owned(w_id uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from public.work_orders w
                 where w.id = w_id and public.zt_is_owner(w.company_id));
$$;
create or replace function public.zt_wo_open(w_id uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from public.work_orders w
                 where w.id = w_id and w.status not in ('done','canceled'));
$$;
grant execute on function public.zt_wo_is_mine(uuid), public.zt_wo_is_owned(uuid), public.zt_wo_open(uuid) to authenticated;
