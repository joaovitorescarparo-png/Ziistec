-- =====================================================================
-- helpers de autorização — SECURITY DEFINER para não recursar em RLS
-- =====================================================================
create or replace function public.zt_is_member(target uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.company_members m
    where m.company_id = target and m.user_id = auth.uid() and m.status = 'active'
  );
$$;

create or replace function public.zt_is_owner(target uuid)
returns boolean language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.company_members m
    where m.company_id = target and m.user_id = auth.uid()
      and m.status = 'active' and m.role = 'owner'
  );
$$;

create or replace function public.zt_is_platform_admin()
returns boolean language sql stable security definer set search_path = public as $$
  select coalesce((select p.is_platform_admin from public.profiles p where p.id = auth.uid()), false);
$$;

  id             uuid primary key default gen_random_uuid(),
  company_id     uuid not null references public.companies(id) on delete cascade,
  number         text not null,
  client_id      uuid not null references public.clients(id) on delete restrict,
  quote_id       uuid references public.quotes(id) on delete set null,
  assigned_to    uuid references public.profiles(id) on delete set null,
  status         zt_wo_status not null default 'unscheduled',
  scheduled_date date,
  scheduled_time time,
  address        text,
  service_place  text,
  request        text,          -- "o que precisa ser feito"
  pre_notes      text,          -- observações antes do atendimento
  pending_note   text,
  extra_cost     numeric(12,2) not null default 0,   -- CUSTO interno (margem). Nunca é cobrado do cliente.

  warranty_id    uuid,
  origin_wo_id   uuid references public.work_orders(id) on delete set null,
  is_warranty_visit boolean not null default false,
  problem_report text,
  billing_entry_id uuid,        -- trava de idempotência: uma OS, uma cobrança
  completed_at   timestamptz,
  created_by     uuid references public.profiles(id),
  created_at     timestamptz not null default now(),
  updated_at     timestamptz not null default now(),
  unique (company_id, number)
);
create index if not exists idx_wo_company   on public.work_orders(company_id, status);
create index if not exists idx_wo_assigned  on public.work_orders(assigned_to, scheduled_date);
create index if not exists idx_wo_date      on public.work_orders(company_id, scheduled_date);
create index if not exists idx_wo_client    on public.work_orders(client_id);
